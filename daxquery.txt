AVERAGEX (
    VALUES ( 'Attendance - Attendance csv'[StudentID] ),
    [Attendance %]
)

AVERAGEX (
    VALUES('Scores - Scores csv'[StudentID]),
    [% Score Per Student]
)

8.

High Performers = 
CALCULATE (
    DISTINCTCOUNT('Scores - Scores csv'[StudentID]),
    FILTER (
        VALUES('Scores - Scores csv'[StudentID]),
        [% Score Per Student] >= 50
    )
)


9.

    TRUE(),
    [% Score] >= 80, "High",
    [% Score] >= 50, "Medium",
    "Low"
)
